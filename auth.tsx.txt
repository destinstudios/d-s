import {createContext,useContext,useEffect,useState} from 'react';import type {ReactNode} from 'react';import {onAuthStateChanged,signInWithEmailAndPassword,signOut,createUserWithEmailAndPassword,sendPasswordResetEmail} from 'firebase/auth';import {doc,getDoc,setDoc,serverTimestamp} from 'firebase/firestore';import {auth,db,firebaseConfigured} from './firebase';
type User={uid:string;email:string|null;role:'admin'|'client';clientId?:string;displayName?:string};type Ctx={user:User|null;loading:boolean;login:(e:string,p:string,role:'admin'|'client')=>Promise<void>;logout:()=>Promise<void>};
const C=createContext<Ctx>(null!);export const useAuth=()=>useContext(C);
export function AuthProvider({children}:{children:ReactNode}){const [user,setUser]=useState<User|null>(null);const [loading,setLoading]=useState(true);
useEffect(()=>{if(!auth||!db){setLoading(false);return;}return onAuthStateChanged(auth,async u=>{try{if(!u){setUser(null);return;}const s=await getDoc(doc(db,'users',u.uid));if(!s.exists()){await signOut(auth);setUser(null);return;}const d=s.data();if(d.active===false){await signOut(auth);setUser(null);return;}setUser({uid:u.uid,email:u.email,role:d.role,clientId:d.clientId,displayName:d.displayName});}catch{setUser(null)}finally{setLoading(false)}})},[]);
const login=async(email:string,password:string,role:'admin'|'client')=>{if(!auth||!db)throw new Error('Firebase setup missing. Add your Firebase web config first.');const cred=await signInWithEmailAndPassword(auth,email,password);const s=await getDoc(doc(db,'users',cred.user.uid));if(!s.exists()||s.data().role!==role||s.data().active===false){await signOut(auth);throw new Error('This account does not have access to this portal.');}};
const logout=async()=>{if(auth)await signOut(auth)};return <C.Provider value={{user,loading,login,logout}}>{children}</C.Provider>}
export async function createClientAccount(email:string,password:string,displayName:string,clientId:string){if(!auth||!db)throw new Error('Firebase setup missing');const c=await createUserWithEmailAndPassword(auth,email,password);await setDoc(doc(db,'users',c.user.uid),{email,displayName,role:'client',clientId,active:true,createdAt:serverTimestamp()});return c.user.uid}
export async function resetPassword(email:string){if(!auth)throw new Error('Firebase setup missing');await sendPasswordResetEmail(auth,email)}
