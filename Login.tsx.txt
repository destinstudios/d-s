import {useState} from 'react';import {useNavigate} from 'react-router-dom';import {useAuth,resetPassword} from '../lib/auth';import {firebaseConfigured} from '../lib/firebase';import {LockKeyhole,Building2} from 'lucide-react';
export default function Login({role}:{role:'admin'|'client'}){const [email,setEmail]=useState('');const [password,setPassword]=useState('');const [error,setError]=useState('');const [busy,setBusy]=useState(false);const {login}=useAuth();const nav=useNavigate();const submit=async(e:React.FormEvent)=>{e.preventDefault();setError('');setBusy(true);try{await login(email,password,role);nav(role==='admin'?'/admin':'/client')}catch(x:any){setError(x.message||'Login failed')}finally{setBusy(false)}};return <main className="login-screen"><form className="login-card" onSubmit={submit}><div className="brand-mark"><Building2 size={26}/></div><div className="eyebrow">DESTIN STUDIOS</div><h1>{role==='admin'?'Admin Login':'Client Portal'}</h1><p>Manage. Grow. Scale.</p>{!firebaseConfigured&&<div className="warning">Firebase web configuration is missing. Add your Firebase values in GitHub repository settings before login.</div>}<label>Email address<input type="email" required value={email} onChange={e=>setEmail(e.target.value)} placeholder="you@example.com"/></label><label>Password<input type="password" required value={password} onChange={e=>setPassword(e.target.value)} placeholder="Your password"/></label>{error&&<div className="error">{error}</div>}<button className="primary full" disabled={busy||!firebaseConfigured}><LockKeyhole size={16}/>{busy?'Signing in...':'Sign in'}</button><button type="button" className="text-btn" onClick={async()=>{if(!email)return setError('Enter your email first');try{await resetPassword(email);setError('Password reset email sent.')}catch(e:any){setError(e.message)}}}>Forgot password?</button><div className="login-links"><a href="#/admin/login">Admin</a><span>·</span><a href="#/client/login">Client</a></div></form></main>}
