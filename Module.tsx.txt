import {useEffect,useState} from 'react';import {addRow,editRow,removeRow,subscribeRows,type Row} from '../lib/data';import {date,money} from '../lib/format';import {Plus,Search,Trash2,Pencil,RefreshCw} from 'lucide-react';
const configs:Record<string,{title:string;collection:string;fields:string[];labels:string[];columns:string[]}>={
Leads:{title:'Leads',collection:'leads',fields:['name','phone','email','business','source','status','nextFollowUp','notes'],labels:['Name','Phone','Email','Business','Source','Status','Next follow-up (YYYY-MM-DD)','Notes'],columns:['name','phone','business','status','source','nextFollowUp']},
Clients:{title:'Clients',collection:'clients',fields:['name','phone','email','business','city','status','notes'],labels:['Name','Phone','Email','Business','City','Status','Notes'],columns:['name','phone','business','city','status']},
Projects:{title:'Projects',collection:'projects',fields:['name','clientId','service','status','progress','startDate','deadline','value','notes'],labels:['Project name','Client ID','Service','Status','Progress %','Start date','Deadline','Project value ₹','Notes'],columns:['name','service','status','progress','deadline','value']},
Tasks:{title:'Tasks',collection:'tasks',fields:['title','assignedTo','dueDate','priority','status','projectId','notes'],labels:['Task','Assigned to','Due date','Priority','Status','Project ID','Notes'],columns:['title','dueDate','priority','status']},
Transactions:{title:'Transactions',collection:'transactions',fields:['type','description','amount','date','clientId','projectId','method','notes'],labels:['Type (income/expense)','Description','Amount ₹','Date','Client ID','Project ID','Method','Notes'],columns:['type','description','amount','date','method']},
Invoices:{title:'Invoices',collection:'invoices',fields:['invoiceNumber','clientId','projectId','total','paid','dueDate','status','notes'],labels:['Invoice number','Client ID','Project ID','Total ₹','Paid ₹','Due date','Status','Notes'],columns:['invoiceNumber','clientId','total','paid','dueDate','status']},
Referrals:{title:'Referrals',collection:'referrals',fields:['code','referrerName','referredName','phone','status','reward','notes'],labels:['Referral code','Referrer','Referred person','Phone','Status','Reward ₹','Notes'],columns:['code','referrerName','referredName','status','reward']},
Support:{title:'Support Tickets',collection:'tickets',fields:['subject','clientId','priority','status','description','resolution'],labels:['Subject','Client ID','Priority','Status','Description','Resolution'],columns:['subject','clientId','priority','status']},
Proposals:{title:'Proposals',collection:'proposals',fields:['title','clientId','amount','status','validUntil','scope'],labels:['Title','Client ID','Amount ₹','Status','Valid until','Scope'],columns:['title','clientId','amount','status','validUntil']},
Campaigns:{title:'Campaigns',collection:'campaigns',fields:['name','channel','startDate','endDate','budget','status','notes'],labels:['Campaign','Channel','Start date','End date','Budget ₹','Status','Notes'],columns:['name','channel','startDate','endDate','budget','status']}
};
export default function Module({name,clientId}:{name:string;clientId?:string}){const c=configs[name];const [rows,setRows]=useState<Row[]>([]);const [search,setSearch]=useState('');const [busy,setBusy]=useState(false);const [error,setError]=useState('');useEffect(()=>subscribeRows(c.collection,setRows,clientId),[c.collection,clientId]);
const create=async()=>{const data:Record<string,any>={};for(let i=0;i<c.fields.length;i++){const v=window.prompt(c.labels[i]);if(v===null)return;if(v.trim())data[c.fields[i]]=['progress','value','amount','total','paid','reward','budget'].includes(c.fields[i])?Number(v):v;}if(clientId)data.clientId=clientId;try{setBusy(true);await addRow(c.collection,data)}catch(e:any){setError(e.message||'Could not save record')}finally{setBusy(false)}};
const update=async(r:Row)=>{const field=c.fields[0];const v=window.prompt(`Update ${c.labels[0]}`,r[field]||'');if(v===null)return;try{await editRow(c.collection,r.id,{[field]:v})}catch(e:any){setError(e.message)}};
const del=async(r:Row)=>{if(!window.confirm('Permanently delete this record? For old leads, prefer changing status to Archived.'))return;try{await removeRow(c.collection,r.id)}catch(e:any){setError(e.message)}};
const filtered=rows.filter(r=>JSON.stringify(r).toLowerCase().includes(search.toLowerCase()));return <section className="panel"><div className="panel-head"><div><h2>{c.title}</h2><p>{rows.length} records · Live database sync</p></div>{(!clientId||name==='Support')&&<button className="primary" onClick={create} disabled={busy}><Plus size={17}/> Add {name.slice(0,-1)||'record'}</button>}</div><div className="toolbar"><div className="search"><Search size={17}/><input placeholder={`Search ${name.toLowerCase()}...`} value={search} onChange={e=>setSearch(e.target.value)}/></div><button className="icon-btn" onClick={()=>setSearch('')} title="Clear search"><RefreshCw size={16}/></button></div>{error&&<div className="error">{error}</div>}<div className="table-wrap"><table><thead><tr>{c.columns.map(k=><th key={k}>{c.labels[c.fields.indexOf(k)]||k}</th>)}{!clientId&&<th>Actions</th>}</tr></thead><tbody>{filtered.map(r=><tr key={r.id}>{c.columns.map(k=><td key={k}>{k.toLowerCase().includes('date')||k==='deadline'||k==='nextFollowUp'?date(r[k]):['amount','total','paid','value','reward','budget'].includes(k)?money(r[k]):k==='progress'?`${r[k]||0}%`:r[k]||'—'}</td>)}{!clientId&&<td><div className="actions"><button className="icon-btn" onClick={()=>update(r)} title="Edit"><Pencil size={15}/></button><button className="icon-btn danger" onClick={()=>del(r)} title="Delete"><Trash2 size={15}/></button></div></td>}</tr>)}{!filtered.length&&<tr><td colSpan={c.columns.length+1} className="empty">{rows.length?'No matching records':'No records yet. Add your first record.'}</td></tr>}</tbody></table></div></section>}
